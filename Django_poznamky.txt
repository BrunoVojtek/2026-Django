8. vytvorenie urlpattern
8.1 nazov_projektu/urls.py
	from django.contrib import admin
	from django.urls import path, include

	urlpatterns = [
    	path('admin/', admin.site.urls),
    	path('skola/', include('nazov_aplikacie.urls')),
	]
8.2 vytvorit subor nazov_aplikacie/urls.py
	from django.urls import path
	from . import views

	urlpatterns = [
    	path('',views.index, name='index'), #views.inex - index = nazov funkcie vo views
	]
